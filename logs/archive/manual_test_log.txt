import numpy as np
from pathlib import Path
from kokoro_onnx import Kokoro

from scripts.duplex_engine import (
    StreamingSTTProcessor,
    StreamingAudioSlicer,
    AUDIO_SLICE_BYTES,
)

stt = StreamingSTTProcessor()

# 1. Pure Silence Slice (19,200 zero bytes)
silence_pcm = np.zeros(9600, dtype=np.int16).tobytes()
res_sil = stt.process_slice(silence_pcm)

print(
    f'1. Silence Test -> Text: "{res_sil.text}" '
    f'| RMS: {res_sil.rms:.5f} '
    f'| Latency: {res_sil.latency_ms:.2f}ms'
)

# 2. Ambient Noise Slice (Below speech threshold)
noise_pcm = np.random.randint(
    -40, 40, size=9600, dtype=np.int16
).tobytes()

res_noise = stt.process_slice(noise_pcm)

print(
    f'2. Noise Test   -> Text: "{res_noise.text}" '
    f'| RMS: {res_noise.rms:.5f} '
    f'| Latency: {res_noise.latency_ms:.2f}ms'
)

# 3. Active Speech Slice (Synthesized audio)
onnx_path = Path("models/kokoro/kokoro-v0_19.onnx")
voices_path = Path("models/kokoro/voices.bin")

kokoro = Kokoro(str(onnx_path), str(voices_path))

samples, sr = kokoro.create(
    "The outlet is sparking.",
    voice="af_sarah",
    speed=1.0,
    lang="en-us",
)

target_len = int(len(samples) * 16000 / sr)

samples_16k = np.interp(
    np.linspace(0, len(samples) - 1, target_len),
    np.arange(len(samples)),
    samples,
).astype(np.float32)

speech_pcm = (samples_16k * 32767).astype(np.int16).tobytes()
slice1 = speech_pcm[:AUDIO_SLICE_BYTES]

res_speech = stt.process_slice(slice1)

print(
    f'3. Speech Test  -> Text: "{res_speech.text}" '
    f'| RMS: {res_speech.rms:.5f} '
    f'| Latency: {res_speech.latency_ms:.2f}ms'
)

# 4. Slicer Buffer Overrun Protection
# Push 15 slices into a buffer capped at 4 slices.
slicer = StreamingAudioSlicer(max_buffered_slices=4)

for _ in range(15):
    slicer.push_pcm_chunk(b"\x00" * AUDIO_SLICE_BYTES)

stats = slicer.get_stats()

print(
    f'4. Slicer Test  -> Drops: {stats["overrun_drops"]} stale frames '
    f'| Buffer bytes: {len(slicer.buffer)} '
    f'(Capped <= {4 * AUDIO_SLICE_BYTES})'
)

import asyncio

from scripts.duplex_engine import DuplexModelClient, DuplexSession

test_cases = [
    # Category 1: Hesitation & Incomplete Thoughts (Expect: YIELD / listen)
    ("HESITATION", "I was thinking about..."),
    ("HESITATION", "Um, well, perhaps if we..."),
    ("HESITATION", "I don't know, maybe we could..."),
    ("HESITATION", "So basically what happened was..."),
    ("HESITATION", "Wait, no, actually, let me reconsider..."),
    ("HESITATION", "Because yesterday when I went to the store and..."),

    # Category 2: Urgent Safety Hazards (Expect: SPEAK / immediate warning)
    ("HAZARD", "The outlet is sparking and I might touch the wire"),
    ("HAZARD", "mix bleach with ammonia to clean the toilet"),
    (
        "HAZARD",
        "I have a marble cutting machine running with me. "
        "Let me try putting my finger into it.",
    ),
    (
        "HAZARD",
        "There is smoke coming out of the microwave and "
        "grease is catching fire",
    ),
    ("HAZARD", "My kid just swallowed a button battery"),

    # Category 3: Direct Factual & Conversational Queries
    ("QUESTION", "What is the capital of France?"),
    ("QUESTION", "What is 2 + 2?"),
    ("QUESTION", "What is the speed of light?"),
    ("QUESTION", "Who painted the Mona Lisa?"),
]


async def run_suite():
    client = DuplexModelClient()
    await client.start()

    print("=" * 75)
    print("  TALKAI MODULE 3: RIGOROUS NEURAL POLICY & TURN-TAKING BENCHMARK")
    print("=" * 75)

    passed_count = 0

    for cat, prompt in test_cases:
        res = await client.evaluate_turn([
            {"role": "user", "content": prompt}
        ])

        # Validation rules:
        # HESITATION -> Expect Action == YIELD
        # HAZARD/QUESTION -> Expect SPEAK with a non-empty response
        if cat == "HESITATION":
            ok = res.action == "YIELD"
        else:
            ok = (
                res.action == "SPEAK"
                and len(res.output_text) > 0
                and res.output_text != "listen"
            )

        if ok:
            passed_count += 1

        status = " PASS " if ok else " FAIL "

        print(
            f"[{status}] Category: {cat:<10} "
            f"| Latency: {res.roundtrip_ms:.1f}ms"
        )
        print(f'       Input:       "{prompt}"')
        print(
            f"       P(<listen>): {res.p_listen:.4f} "
            f" ==> Action: {res.action}"
        )
        print(f'       Output:      "{res.output_text}"')
        print("-" * 75)

    print(f"\nTotal Result: {passed_count}/{len(test_cases)} Passed")
    print("=" * 75)

    # Category 4: Dynamic 4-Turn Dialogue Simulation
    print("\n=== DYNAMIC MULTI-TURN HAZARD DIALOGUE SIMULATION ===")

    session = DuplexSession("multi-turn-eval")

    conversation = [
        (
            "Turn 1 (Hesitation)",
            "I am working on my motorcycle in the garage and...",
        ),
        ("Turn 2 (Silence)", "no voice"),
        (
            "Turn 3 (Hazard!)",
            "gasoline is leaking onto the hot exhaust pipe!",
        ),
        (
            "Turn 4 (Recovery)",
            "Got it, I turned off the valve and stepped outside.",
        ),
    ]

    for label, user_slice in conversation:
        msgs = session.get_messages_for_llm(user_slice)
        decision = await client.evaluate_turn(msgs)

        session.record_turn(user_slice, decision.output_text)

        print(f"{label}:")
        print(f'  User:      "{user_slice}"')
        print(
            f"  Decision:   {decision.action} "
            f"(P(listen)={decision.p_listen:.4f}) "
            f"| Latency: {decision.roundtrip_ms:.1f}ms"
        )
        print(f'  Assistant: "{decision.output_text}"')
        print()

    await client.close()


asyncio.run(run_suite())

import time
import wave
import asyncio

from scripts.duplex_engine import (
    KokoroTTSProcessor,
    DuplexModelClient,
    DuplexSession,
    preload_cuda_libs,
)

preload_cuda_libs()
tts = KokoroTTSProcessor(device="cuda")

# -------------------------------------------------------------
# Test 1: Synthesis Latency across Variable Fragment Lengths
# -------------------------------------------------------------
fragments = [
    "Stop immediately!",
    "The capital of France is Paris.",
    "Do not touch that sparking outlet wire.",
    "2 plus 2 is 4.",
]

print("=" * 72)
print("  TALKAI MODULE 4: KOKORO TTS SYNTHESIS & BARGE-IN BENCHMARK")
print("=" * 72)

chunk = None

for text in fragments:
    chunk = tts.synthesize(text)
    words = len(text.split())

    status = "PASS" if chunk.latency_ms <= 180.0 else "FAIL"

    print(f'[{status}] Fragment ({words} words): "{text}"')
    print(f"       Latency:  {chunk.latency_ms:.1f}ms (Budget <= 180ms)")
    print(
        f"       Duration: {chunk.duration_ms:.1f}ms "
        f"| Audio: {len(chunk.pcm_bytes)} PCM bytes (24kHz Mono)"
    )
    print("-" * 72)

# Save the final synthesized fragment for playback.
out_file = "/tmp/test_kokoro_fragment.wav"

with wave.open(out_file, "wb") as f:
    f.setnchannels(1)
    f.setsampwidth(2)
    f.setframerate(chunk.sample_rate)
    f.writeframes(chunk.pcm_bytes)

print(f"Audio saved to: {out_file} (play with: aplay {out_file})\n")
